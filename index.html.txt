        // 2. Render Athlete Cards Grid
        function renderAthletes() {
            const grid = document.getElementById('athletes-grid');
            const searchQuery = document.getElementById('search-input').value.toLowerCase().trim();
            const sortBy = document.getElementById('sort-select').value;

            let filtered = athletes.filter(a => {
                const matchesSport = currentSportFilter === 'All Sports' || a.sport === currentSportFilter;
                const matchesSearch = !searchQuery ||
                    a.name.toLowerCase().includes(searchQuery) ||
                    a.university.toLowerCase().includes(searchQuery) ||
                    a.sport.toLowerCase().includes(searchQuery);
                return matchesSport && matchesSearch;
            });

            // Sorting logic
            filtered.sort((a, b) => {
                if (sortBy === 'closest') {
                    return (b.raised / b.goal) - (a.raised / a.goal);
                } else if (sortBy === 'lowest') {
                    return (a.goal - a.raised) - (b.goal - b.raised);
                } else {
                    return b.createdAt - a.createdAt;
                }
            });

            if (filtered.length === 0) {
                grid.innerHTML = `
                    <div class="col-span-full bg-white rounded-3xl p-12 text-center border border-slate-200">
                        <p class="font-display font-bold text-lg text-navy-900">No matching athletes found</p>
                        <p class="text-sm text-slate-500 mt-1">Try resetting the sport filter or search term.</p>
                        <button onclick="setSportFilter('All Sports'); document.getElementById('search-input').value=''; renderAthletes();" class="mt-4 px-5 py-2.5 rounded-xl bg-navy-900 text-white text-sm font-bold">Reset Filters</button>
                    </div>
                `;
                return;

                const pct = Math.min(100, Math.round((athlete.raised / athlete.goal) * 100));

                return `

        // 4. Micro-Sponsorship Checkout Modal Functions
        function openSponsorModal(athleteId) {
            const athlete = athletes.find(a => a.id === athleteId);
            if (!athlete) return;
            activeSponsorAthleteId = athleteId;

            document.getElementById('modal-athlete-name').textContent = `Sponsor ${athlete.name}`;
            document.getElementById('modal-athlete-meta').textContent = `${athlete.university} • ${athlete.sport}`;
            updateModalProgress(athlete);

            selectTier(2500);
            const modal = document.getElementById('sponsor-modal');
            modal.classList.remove('hidden');
            modal.classList.add('flex');
        }

        function updateModalProgress(athlete) {
            const pct = Math.min(100, Math.round((athlete.raised / athlete.goal) * 100));
            document.getElementById('modal-raised-label').textContent = `₹${athlete.raised.toLocaleString('en-IN')} raised (${pct}%)`;
            document.getElementById('modal-goal-label').textContent = `Goal: ₹${athlete.goal.toLocaleString('en-IN')}`;
            document.getElementById('modal-progress-bar').style.width = `${pct}%`;
        }

        function closeSponsorModal() {
            const modal = document.getElementById('sponsor-modal');
            modal.classList.add('hidden');
            modal.classList.remove('flex');
            activeSponsorAthleteId = null;
        }

        function selectTier(amount) {
            document.getElementById('sponsor-amount').value = amount;
            document.querySelectorAll('.tier-btn').forEach(btn => {
                if (Number(btn.dataset.amount) === amount) {
                    btn.className = 'tier-btn p-3 rounded-xl border-2 border-emerald-500 bg-emerald-500/10 text-center transition-all';
                } else {
                    btn.className = 'tier-btn p-3 rounded-xl border-2 border-slate-200 hover:border-emerald-500 text-center transition-all';
                }
            });
        }

        function submitSponsorship(event) {
            event.preventDefault();
            const amount = Number(document.getElementById('sponsor-amount').value);
            const brand = document.getElementById('sponsor-brand').value.trim();
            const athlete = athletes.find(a => a.id === activeSponsorAthleteId);

            if (!athlete || amount < 100 || !brand) return;

            // Update live data
            athlete.raised += amount;
            if (!athlete.sponsors.includes(brand)) {
                athlete.sponsors.push(brand);
            }

            renderAthletes();
            closeSponsorModal();
            document.getElementById('sponsor-brand').value = '';

            showToast(`Thank you, ${brand}! ₹${amount.toLocaleString('en-IN')} pledged to ${athlete.name}.`);
        }
